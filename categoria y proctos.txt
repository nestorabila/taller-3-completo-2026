@Component({

      imports: [NgxPaginationModule, MessageModule,MessagesModule, ConfirmDialog, ReactiveFormsModule, FormsModule, InputTextModule, Dialog, CommonModule, ConfirmPopupModule, ToastModule, ButtonModule],
    providers: [ConfirmationService, MessageService],

  styleUrl: './categoria.component.css'
})
export class CategoriaComponent {
  pag=1;
   mostrarFormulario: boolean = false;
    Obtenercategoria!: Categoria;
  vista(fall: Categoria) {
      this.Obtenercategoria = fall; 
      this.modelSubCategoria(); 
       this.radioSeleccionado = true;
    }
  xper!: any;
  xcate!: any
   listaSubCategoria: Subcategoria[]=[];
   listaCategoria: Categoria[]=[];

  subcateForm: any = new FormGroup({
      idsubc: new FormControl(''),
      nombre: new FormControl(''),
      descripcion: new FormControl(''),
      categoria: new FormControl(''),
    })
    cateForm: any = new FormGroup({
      idcategoria: new FormControl(''),
      nombre: new FormControl(''),
      hora: new FormControl(''),
      fecha: new FormControl(''),
    })
     

  constructor(private cateService: CategoriaService, private confirmationService: ConfirmationService, private messageService: MessageService) {}

modelCategoria() {
  this.cateService.ListarCategoria().subscribe((fall) => {
    this.listaCategoria = fall;
  });
}
modelSubCategoria() {
  if (this.Obtenercategoria) {
    this.cateService.ListarSubCategoriaID(this.Obtenercategoria.idcategoria).subscribe((fall) => {
      this.listaSubCategoria
     = fall;
      console.log('negocios obtenidas para el cliente:', this.Obtenercategoria.nombre, fall);
    });
  } else {
    console.warn('No hay cliente seleccionado para cargar negocio');
  }
}
  ngOnInit(): void {
    this.modelCategoria();
    this.modelSubCategoria();
  }
 filtro: string = ''; 
  buscarCategoria() {
    if (this.filtro.trim()) {
      this.cateService.buscarCategorias(this.filtro).subscribe((result) => {
        this.listaCategoria = result;
      });
    } else {
      this.modelCategoria(); 
    }

  }
 

registrarCategoria() {
    const fecha = new Date();
    const hora = fecha.toTimeString().split(' ')[0];
    this.xper = {
      nombre: this.cateForm.get('nombre')?.value,
      fecha: fecha,
      hora: hora
    },
    this.cateService
      .RegistrarCategoria(this.xper)
      .subscribe(() => {
        this.cateService.ListarCategoria().subscribe((fall) => {
    this.listaCategoria = fall;
    console.log("Lista actualizada:", this.listaCategoria);
  });
  
      });
  }

// modificar categoria
vistaCategoria(xdat: Categoria) { 
  this.cateForm.patchValue({
    idcategoria: xdat.idcategoria,
    fecha: xdat.fecha,
    hora: xdat.hora,
    nombre: xdat.nombre
  })
}
modificarCategoria() {
    this.xper = {
      idcategoria: this.cateForm.get('idcategoria')?.value,
      nombre: this.cateForm.get('nombre')?.value,
      fecha: this.cateForm.get('fecha')?.value,
      hora: this.cateForm.get('hora')?.value,
    },
    this.cateService
      .ModificarCategoria(this.xper)
      .subscribe(() => {
        this.cateService.ListarCategoria().subscribe((fall) => {
    this.listaCategoria = fall;
    console.log("Lista actualizada:", this.listaCategoria);
  });
  
      });
  }
//  registrar sub categoria
  registrarSubCategoria() {
    const categoria = this.Obtenercategoria;
    this.xcate = {
      nombre: this.subcateForm.get('nombre')?.value,
      descripcion: this.subcateForm.get('descripcion')?.value,
      categoria: categoria,
    },
     console.log("datos de de subcategoria mandando::",this.xcate);

    this.cateService
      .RegistrarSubCategoria(this.xcate)
      .subscribe(() => {
        this.cateService.ListarSubCategoriaID(this.Obtenercategoria.idcategoria).subscribe((fall) => (this.listaSubCategoria=fall))
        this.subcateForm.reset();
      });
  }
// modificar sub categoria
vistaSubCategoria(xdat: Subcategoria) { 
  this.subcateForm.patchValue({
    idsubc: xdat.idsubc,
    descripcion: xdat.descripcion,
    categoria: xdat.categoria,
    nombre: xdat.nombre
  })
}
modificarSubCategoria() {
    const categoria = this.Obtenercategoria;
    this.xcate = {
      idsubc: this.subcateForm.get('idsubc')?.value,
      nombre: this.subcateForm.get('nombre')?.value,
      descripcion: this.subcateForm.get('descripcion')?.value,
      categoria: this.subcateForm.get('categoria')?.value,
    },
     console.log("datos de de subcategoria mandando::",this.xcate);

    this.cateService
      .ModificarSubCategoria(this.xcate)
      .subscribe(() => {
        this.cateService.ListarSubCategoriaID(this.Obtenercategoria.idcategoria).subscribe((fall) => (this.listaSubCategoria=fall))
        this.subcateForm.reset();
      });
  }












perInfo = '';
  eliminarNegocio!: Subcategoria;
  vistaEliminar(fall: Subcategoria ){
    this.perInfo = fall.nombre;
    this.eliminarNegocio = fall;
    
  }
  EliminarSubCategoria() {
    if (this.eliminarNegocio && this.eliminarNegocio.idsubc) {
      this.cateService.eliminarSubCategoria(this.eliminarNegocio.idsubc).subscribe({
        next: () => {
          this.cateService.ListarSubCategoriaID(this.Obtenercategoria.idcategoria).subscribe((fall) => (this.listaSubCategoria=fall));
          this.messageService.add({
          severity: 'success',
          summary: '¡Eliminación exitosa!',
          detail: `${this.perInfo} fue eliminado correctamente.`,
          life: 3000
        });
        },
         error: (err) => {
        console.error('Error al eliminar Producto:', err);
        this.messageService.add({
          severity: 'error',
          summary: 'Error al eliminar',
          detail: 'No se puede eliminar este producto porque ya está siendo usado.',
          life: 4000
        });
      }
      });
    }
  }

  perInfo1 = '';
  eliminarCategoria!: Categoria;
  vistaEliminar1(fall: Categoria ){

  confirm2(event: Event) {
        this.confirmationService.confirm({
            target: event.target as EventTarget,

            header: 'Confirmar Eliminacion',

            acceptButtonProps: {

validarNombre(): void {
  const nombreControl = this.cateForm.get('nombre');
  const valor: string = nombreControl?.value || '';
// 5. Ya existe una categoría con ese nombre
const yaExiste = this.listaCategoria.some(cat =>
  cat.nombre.trim().toLowerCase() === valor.trim().toLowerCase()
);
if (yaExiste) {
  this.mostrarMensaje('El nombre ya fue usado');
  this.botonHabilitado = false;
  return;
}

  // 1. Campo vacío
  if (!valor.trim()) {
    this.mostrarMensaje('El campo nombre es requerido');
    this.botonHabilitado = false;
    return;
  }

  // 2. Caracteres inválidos
  const soloLetrasEspacio = /^[a-zA-ZáéíóúÁÉÍÓÚñÑ ]+$/;
  if (!soloLetrasEspacio.test(valor)) {
    this.mostrarMensaje('Formato no permitido');
    this.botonHabilitado = false;
    return;
  }

  // 3. Mínimo 3 letras
  if (valor.trim().length < 3) {
    this.mostrarMensaje('Debe contener al menos 3 caracteres');
    this.botonHabilitado = false;
    return;
  }

  // 4. Máximo 50 letras
  if (valor.trim().length > 50) {
    this.mostrarMensaje('No debe superar los 50 caracteres');
    this.botonHabilitado = false;
    return;
  }

  // ✅ Todo válido
  this.mensajeVisible = false;
  this.botonHabilitado = true;
}
